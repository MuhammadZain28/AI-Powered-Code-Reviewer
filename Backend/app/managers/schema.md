```sql
CREATE SCHEMA "public";
CREATE SCHEMA "auth";
CREATE SCHEMA "pgrst";
CREATE TABLE "calls" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"caller_id" uuid NOT NULL,
	"function_name" text,
	"call_type" varchar(50),
	"created_at" timestamp DEFAULT now(),
	"resolve_to" text,
	"library" text,
	"source" text,
	"callee_id" uuid
);
CREATE TABLE "chunks" (
	"id" uuid PRIMARY KEY,
	"chunk_type" text,
	"name" text,
	"start_line" integer,
	"end_line" integer,
	"content" text,
	"created_at" timestamp DEFAULT now(),
	"docstring" text,
	"parameters" text[],
	"return_values" text[],
	"complexity" integer DEFAULT 1,
	"class_id" uuid,
	"file_id" uuid,
	"hash" text,
	"class_name" text,
	"model" text DEFAULT 'all-MiniLM-L6-v2',
	"embedding_id" bigint GENERATED BY DEFAULT AS IDENTITY (sequence name "chunks_embedding_id_seq" INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START WITH 1 CACHE 1),
	"score" numeric
);
CREATE TABLE "class_attributes" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"class_id" uuid NOT NULL,
	"name" text NOT NULL,
	"attribute_type" text,
	"default_value" text,
	"is_static" boolean DEFAULT false,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "classes" (
	"id" uuid PRIMARY KEY,
	"file_id" uuid NOT NULL,
	"name" text NOT NULL,
	"start_line" integer,
	"end_line" integer,
	"docstring" text,
	"inheritance" text[],
	"created_at" timestamp DEFAULT now(),
	"hash" text
);
CREATE TABLE "files" (
	"id" uuid PRIMARY KEY,
	"project_id" uuid,
	"path" text NOT NULL CONSTRAINT "files_path_key" UNIQUE,
	"language" text,
	"hash" text,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "import_modules" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"import_id" uuid,
	"module" text,
	"alias" text
);
CREATE TABLE "imports" (
	"id" uuid PRIMARY KEY,
	"file_id" uuid NOT NULL,
	"source" text,
	"type" text,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "projects" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"name" text NOT NULL,
	"created_at" timestamp DEFAULT now(),
	"path" text NOT NULL,
	"description" text,
	"features" text[],
	"modules" text[],
	"frontend" text,
	"backend" text,
	"technologies" text[]
);
CREATE TABLE "review_queue" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"chunk_id" uuid NOT NULL CONSTRAINT "review_queue_chunk_id_key" UNIQUE,
	"priority" integer,
	"updated_at" timestamp DEFAULT now(),
	"status" text
);
CREATE TABLE "reviews" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"severity" text,
	"category" text,
	"review" text,
	"suggested_fix" text,
	"created_at" timestamp DEFAULT now(),
	"chunk_id" uuid NOT NULL
);
CREATE TABLE "summaries" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"module" text,
	"purpose" text,
	"chunk_id" uuid
);
CREATE UNIQUE INDEX "calls_pkey" ON "calls" ("id");
CREATE INDEX "idx_calls_caller" ON "calls" ("caller_id");
CREATE INDEX "idx_calls_function" ON "calls" ("function_name");
CREATE UNIQUE INDEX "chunks_pkey" ON "chunks" ("id");
CREATE INDEX "idx_chunks_class" ON "chunks" ("class_id");
CREATE INDEX "idx_chunks_file" ON "chunks" ("file_id");
CREATE INDEX "idx_chunks_name" ON "chunks" ("name");
CREATE UNIQUE INDEX "class_attributes_pkey" ON "class_attributes" ("id");
CREATE INDEX "idx_class_attributes_class" ON "class_attributes" ("class_id");
CREATE UNIQUE INDEX "classes_pkey" ON "classes" ("id");
CREATE INDEX "idx_classes_file" ON "classes" ("file_id");
CREATE UNIQUE INDEX "files_path_key" ON "files" ("path");
CREATE UNIQUE INDEX "files_pkey" ON "files" ("id");
CREATE INDEX "idx_files_project" ON "files" ("project_id");
CREATE INDEX "idx_import_modules_import" ON "import_modules" ("import_id");
CREATE UNIQUE INDEX "import_modules_pkey" ON "import_modules" ("id");
CREATE INDEX "idx_imports_file" ON "imports" ("file_id");
CREATE UNIQUE INDEX "imports_pkey" ON "imports" ("id");
CREATE UNIQUE INDEX "projects_pkey" ON "projects" ("id");
CREATE UNIQUE INDEX "review_queue_chunk_id_key" ON "review_queue" ("chunk_id");
CREATE UNIQUE INDEX "review_queue_pkey" ON "review_queue" ("id");
CREATE UNIQUE INDEX "reviews_pkey" ON "reviews" ("id");
CREATE UNIQUE INDEX "summaries_pkey" ON "summaries" ("id");

ALTER TABLE "classes" ADD CONSTRAINT "classes_file_id_fkey" FOREIGN KEY ("file_id") REFERENCES "files"("id") ON DELETE CASCADE;
ALTER TABLE "files" ADD CONSTRAINT "files_project_id_fkey" FOREIGN KEY ("project_id") REFERENCES "projects"("id") ON DELETE CASCADE;
ALTER TABLE "import_modules" ADD CONSTRAINT "import_modules_import_id_fkey" FOREIGN KEY ("import_id") REFERENCES "imports"("id") ON DELETE CASCADE;
ALTER TABLE "imports" ADD CONSTRAINT "imports_file_id_fkey" FOREIGN KEY ("file_id") REFERENCES "files"("id") ON DELETE CASCADE;
ALTER TABLE "review_queue" ADD CONSTRAINT "review_queue_fkey" FOREIGN KEY ("chunk_id") REFERENCES "chunks"("id") ON DELETE CASCADE;
ALTER TABLE "reviews" ADD CONSTRAINT "fkey_chunk" FOREIGN KEY ("chunk_id") REFERENCES "chunks"("id") ON DELETE CASCADE;
ALTER TABLE "summaries" ADD CONSTRAINT "summary_fkey" FOREIGN KEY ("chunk_id") REFERENCES "chunks"("id") ON DELETE CASCADE;
```
