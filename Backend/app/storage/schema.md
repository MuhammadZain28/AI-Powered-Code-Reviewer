```sql
CREATE SCHEMA "public";
CREATE SCHEMA "auth";
CREATE SCHEMA "pgrst";
CREATE TABLE "call_graph" (
	"id" uuid DEFAULT gen_random_uuid(),
	"caller_id" uuid NOT NULL,
	"callee_id" uuid,
	"function_name" text,
	"call_type" text,
	"call_line" integer,
	"library" text,
	"resolved_path" text,
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "calls_pkey" PRIMARY KEY("id"),
	CONSTRAINT "calls_call_type_check" CHECK ((call_type = ANY (ARRAY['direct'::text, 'conditional'::text, 'async'::text, 'callback'::text, 'recursive'::text]))),
	CONSTRAINT "calls_destination_check" CHECK (((callee_id IS NOT NULL) OR (library IS NOT NULL)))
);
CREATE TABLE "class_attributes" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"class_id" uuid NOT NULL,
	"name" text NOT NULL,
	"attribute_type" text,
	"default_value" text,
	"line_number" integer,
	"is_static" boolean DEFAULT false,
	"is_private" boolean DEFAULT false,
	"is_class_var" boolean DEFAULT false,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "class_inheritance" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"child_class_id" uuid NOT NULL,
	"parent_name" text NOT NULL,
	"parent_class_id" uuid,
	"is_external" boolean DEFAULT false
);
CREATE TABLE "classes" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"file_id" uuid NOT NULL,
	"name" text NOT NULL,
	"start_line" integer,
	"end_line" integer,
	"is_abstract" boolean DEFAULT false,
	"method_count" integer DEFAULT 0,
	"hash" text,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "embeddings" (
	"id" bigint PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY (sequence name "embeddings_id_seq" INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START WITH 1 CACHE 1),
	"chunk_id" uuid NOT NULL UNIQUE,
	"model" text NOT NULL UNIQUE,
	"embedding" vector(384),
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "embeddings_chunk_id_model_key" UNIQUE("chunk_id","model")
);
CREATE TABLE "files" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"project_id" uuid NOT NULL UNIQUE,
	"path" text NOT NULL UNIQUE,
	"language" text,
	"framework" text,
	"total_lines" integer,
	"hash" text,
	"created_at" timestamp DEFAULT now(),
	"updated_at" timestamp DEFAULT now(),
	CONSTRAINT "files_project_path_key" UNIQUE("project_id","path")
);
CREATE TABLE "functions" (
	"id" uuid DEFAULT gen_random_uuid(),
	"file_id" uuid NOT NULL,
	"class_id" uuid,
	"name" text,
	"start_line" integer NOT NULL,
	"end_line" integer NOT NULL,
	"content" text NOT NULL,
	"signature" text,
	"cyclomatic_complexity" integer DEFAULT 1,
	"hash" text,
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "chunks_pkey" PRIMARY KEY("id")
);
CREATE TABLE "import_symbols" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"import_id" uuid NOT NULL,
	"symbol" text,
	"alias" text,
	"is_used" boolean DEFAULT true
);
CREATE TABLE "imports" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"file_id" uuid NOT NULL,
	"from_module" text,
	"type" text,
	"is_external" boolean DEFAULT false,
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "imports_type_check" CHECK ((type = ANY (ARRAY['default'::text, 'named'::text, 'wildcard'::text, 'side_effect'::text])))
);
CREATE TABLE "projects" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"name" text NOT NULL,
	"path" text NOT NULL,
	"description" text,
	"frontend" text,
	"backend" text,
	"features" text[],
	"modules" text[],
	"technologies" text[],
	"created_at" timestamp DEFAULT now(),
	"updated_at" timestamp DEFAULT now()
);
CREATE TABLE "review_queue" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"chunk_id" uuid NOT NULL,
	"review_run_id" uuid,
	"priority" integer,
	"status" text DEFAULT 'pending',
	"error_message" text,
	"started_at" timestamp,
	"completed_at" timestamp,
	"updated_at" timestamp DEFAULT now(),
	CONSTRAINT "review_queue_priority_check" CHECK (((priority >= 1) AND (priority <= 10))),
	CONSTRAINT "review_queue_status_check" CHECK ((status = ANY (ARRAY['pending'::text, 'processing'::text, 'completed'::text, 'failed'::text])))
);
CREATE TABLE "review_runs" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"project_id" uuid,
	"status" text,
	"model" text,
	"started_at" timestamp,
	"completed_at" timestamp,
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "review_runs_status_check" CHECK ((status = ANY (ARRAY['pending'::text, 'processing'::text, 'completed'::text, 'failed'::text])))
);
CREATE TABLE "reviews" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"chunk_id" uuid NOT NULL,
	"review_run_id" uuid,
	"severity" text,
	"category" text,
	"line_start" integer,
	"line_end" integer,
	"review" text,
	"suggested_fix" text,
	"is_resolved" boolean DEFAULT false,
	"created_at" timestamp DEFAULT now(),
	CONSTRAINT "reviews_category_check" CHECK ((category = ANY (ARRAY['security'::text, 'performance'::text, 'style'::text, 'correctness'::text, 'maintainability'::text, 'documentation'::text]))),
	CONSTRAINT "reviews_severity_check" CHECK ((severity = ANY (ARRAY['critical'::text, 'error'::text, 'warning'::text, 'info'::text])))
);
CREATE TABLE "summaries" (
	"id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
	"chunk_id" uuid NOT NULL,
	"purpose" text,
	"model" text,
	"created_at" timestamp DEFAULT now()
);
CREATE UNIQUE INDEX "calls_pkey" ON "call_graph" ("id");

CREATE INDEX "idx_review_queue_status" ON "review_queue" ("status");
```
